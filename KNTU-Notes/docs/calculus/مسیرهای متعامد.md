<script>
(function() {
  const canvas = document.getElementById('orthoCanvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');

  const familySelect = document.getElementById('curveFamilySelect');
  const densitySlider = document.getElementById('densitySlider');
  const densityVal = document.getElementById('densityVal');
  const showTangentsCheck = document.getElementById('showTangentsCheck');
  const tooltip = document.getElementById('orthoTooltip');

  const W = canvas.width;
  const H = canvas.height;
  const originX = W / 2;
  const originY = H / 2;
  const scale = 45; // پیکسل به ازای هر واحد ریاضی

  let mouseX = originX + 2.5 * scale;
  let mouseY = originY - 1.8 * scale;
  let hasHover = false;

  function toScreenX(x) { return originX + x * scale; }
  function toScreenY(y) { return originY - y * scale; }
  function toMathX(px) { return (px - originX) / scale; }
  function toMathY(py) { return -(py - originY) / scale; }

  function drawGrid() {
    ctx.clearRect(0, 0, W, H);

    // محورها
    ctx.lineWidth = 1;
    ctx.strokeStyle = '#1e293b';
    ctx.beginPath();
    for (let x = -10; x <= 10; x++) {
      ctx.moveTo(toScreenX(x), 0);
      ctx.lineTo(toScreenX(x), H);
    }
    for (let y = -8; y <= 8; y++) {
      ctx.moveTo(0, toScreenY(y));
      ctx.lineTo(W, toScreenY(y));
    }
    ctx.stroke();

    // محور مختصات اصلی
    ctx.lineWidth = 1.5;
    ctx.strokeStyle = '#475569';
    ctx.beginPath();
    ctx.moveTo(0, originY); ctx.lineTo(W, originY);
    ctx.moveTo(originX, 0); ctx.lineTo(originX, H);
    ctx.stroke();
  }

  function renderSimulation() {
    drawGrid();

    const family = familySelect.value;
    const count = parseInt(densitySlider.value, 10);
    const showTangents = showTangentsCheck.checked;

    if (family === 'parabolas') {
      // اصلی: y = c * x^2 (سهمی‌ها به رنگ آبی روشن)
      ctx.strokeStyle = '#38bdf8';
      ctx.lineWidth = 2;
      for (let i = 1; i <= count; i++) {
        const c = (i * 0.25);
        // سهمی رو به بالا و پایین
        [c, -c].forEach(valC => {
          ctx.beginPath();
          let started = false;
          for (let px = 0; px <= W; px += 3) {
            const mx = toMathX(px);
            const my = valC * mx * mx;
            const py = toScreenY(my);
            if (!started) { ctx.moveTo(px, py); started = true; }
            else { ctx.lineTo(px, py); }
          }
          ctx.stroke();
        });
      }

      // متعامد: x^2 / (2k) + y^2 / k = 1 (بیضی‌ها به رنگ زرشکی/صورتی نئون)
      ctx.strokeStyle = '#f43f5e';
      ctx.lineWidth = 2;
      for (let i = 1; i <= count; i++) {
        const k = i * 1.4;
        const a = Math.sqrt(2 * k) * scale;
        const b = Math.sqrt(k) * scale;
        ctx.beginPath();
        ctx.ellipse(originX, originY, a, b, 0, 0, Math.PI * 2);
        ctx.stroke();
      }

    } else if (family === 'hyperbolas') {
      // اصلی: x * y = c
      ctx.strokeStyle = '#38bdf8';
      ctx.lineWidth = 2;
      for (let i = 1; i <= count; i++) {
        const c = i * 0.8;
        [c, -c].forEach(valC => {
          ctx.beginPath();
          let started = false;
          for (let px = 0; px <= W; px += 2) {
            const mx = toMathX(px);
            if (Math.abs(mx) < 0.15) { started = false; continue; }
            const my = valC / mx;
            const py = toScreenY(my);
            if (!started) { ctx.moveTo(px, py); started = true; }
            else { ctx.lineTo(px, py); }
          }
          ctx.stroke();
        });
      }

      // متعامد: x^2 - y^2 = k
      ctx.strokeStyle = '#f43f5e';
      ctx.lineWidth = 2;
      for (let i = 1; i <= count; i++) {
        const k = i * 1.5;
        // شاخه‌های چپ و راست
        ctx.beginPath();
        for (let py = 0; py <= H; py += 2) {
          const my = toMathY(py);
          const xSq = k + my * my;
          if (xSq >= 0) {
            const mx = Math.sqrt(xSq);
            ctx.lineTo(toScreenX(mx), py);
          }
        }
        ctx.stroke();
        ctx.beginPath();
        for (let py = 0; py <= H; py += 2) {
          const my = toMathY(py);
          const xSq = k + my * my;
          if (xSq >= 0) {
            const mx = -Math.sqrt(xSq);
            ctx.lineTo(toScreenX(mx), py);
          }
        }
        ctx.stroke();
      }

    } else if (family === 'circles') {
      // اصلی: دایره‌ها x^2 + y^2 = c^2
      ctx.strokeStyle = '#38bdf8';
      ctx.lineWidth = 2;
      for (let i = 1; i <= count; i++) {
        const r = (i * 0.8) * scale;
        ctx.beginPath();
        ctx.arc(originX, originY, r, 0, Math.PI * 2);
        ctx.stroke();
      }

      // متعامد: خطوط مستقیم شعاعی y = k * x
      ctx.strokeStyle = '#f43f5e';
      ctx.lineWidth = 2;
      const angles = count * 2;
      for (let i = 0; i < angles; i++) {
        const th = (i / angles) * Math.PI;
        ctx.beginPath();
        ctx.moveTo(originX - Math.cos(th) * W, originY + Math.sin(th) * W);
        ctx.lineTo(originX + Math.cos(th) * W, originY - Math.sin(th) * W);
        ctx.stroke();
      }
    }

    // ترسیم خطوط مماس متقاطع روی نقطه ماوس
    if (showTangents) {
      const mx = toMathX(mouseX);
      const my = toMathY(mouseY);

      if (Math.abs(mx) > 0.1 || Math.abs(my) > 0.1) {
        let m1 = 0;
        let m2 = 0;

        if (family === 'parabolas') {
          m1 = (2 * my) / (mx || 0.001); // شیب سهمی
          m2 = -mx / (2 * (my || 0.001)); // شیب بیضی
        } else if (family === 'hyperbolas') {
          m1 = -my / (mx || 0.001);
          m2 = mx / (my || 0.001);
        } else if (family === 'circles') {
          m1 = -mx / (my || 0.001);
          m2 = my / (mx || 0.001);
        }

        // خط مماس اول
        const len = 70;
        const ang1 = Math.atan(m1);
        ctx.strokeStyle = '#38bdf8';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(mouseX - Math.cos(ang1) * len, mouseY + Math.sin(ang1) * len);
        ctx.lineTo(mouseX + Math.cos(ang1) * len, mouseY - Math.sin(ang1) * len);
        ctx.stroke();

        // خط مماس دوم (عمود بر اولی)
        const ang2 = Math.atan(m2);
        ctx.strokeStyle = '#f43f5e';
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(mouseX - Math.cos(ang2) * len, mouseY + Math.sin(ang2) * len);
        ctx.lineTo(mouseX + Math.cos(ang2) * len, mouseY - Math.sin(ang2) * len);
        ctx.stroke();

        // نماد زاویه قائمه (Square angle marker)
        const markerSize = 14;
        const u1x = Math.cos(ang1), u1y = -Math.sin(ang1);
        const u2x = Math.cos(ang2), u2y = -Math.sin(ang2);
        ctx.strokeStyle = '#22c55e';
        ctx.lineWidth = 2.5;
        ctx.beginPath();
        ctx.moveTo(mouseX + u1x * markerSize, mouseY + u1y * markerSize);
        ctx.lineTo(mouseX + (u1x + u2x) * markerSize, mouseY + (u1y + u2y) * markerSize);
        ctx.lineTo(mouseX + u2x * markerSize, mouseY + u2y * markerSize);
        ctx.stroke();

        // نقطه تقاطع
        ctx.fillStyle = '#22c55e';
        ctx.beginPath();
        ctx.arc(mouseX, mouseY, 6, 0, Math.PI * 2);
        ctx.fill();

        // متن راهنما
        tooltip.innerHTML = `نقطه: (${mx.toFixed(2)}, ${my.toFixed(2)}) | شیب منحنی ۱: ${m1.toFixed(2)} | شیب منحنی ۲: ${m2.toFixed(2)} | حاصل‌ضرب شیب‌ها: <strong>-۱.۰۰</strong> (تعامد کامل ۹۰°)`;
      }
    }
  }

  // ماوس ایونت‌ها
  canvas.addEventListener('mousemove', e => {
    const rect = canvas.getBoundingClientRect();
    const scaleX = canvas.width / rect.width;
    const scaleY = canvas.height / rect.height;
    mouseX = (e.clientX - rect.left) * scaleX;
    mouseY = (e.clientY - rect.top) * scaleY;
    hasHover = true;
    renderSimulation();
  });

  familySelect.addEventListener('change', renderSimulation);
  densitySlider.addEventListener('input', e => {
    densityVal.innerText = e.target.value;
    renderSimulation();
  });
  showTangentsCheck.addEventListener('change', renderSimulation);

  renderSimulation();
})();
</script>

## مفهوم هندسی و فیزیکی مسیرهای متعامد
در هندسه تحلیلی و معادلات دیفرانسیل، اگر دسته‌ای از منحنی‌ها با رابطه $F(x, y, c) = 0$ مشخص شده باشند، به دسته‌ای از منحنی‌ها مانند $G(x, y, k) = 0$ **مسیرهای متعامد** می‌گویند اگر در هر نقطه تقاطع، مماس‌های دو منحنی بر یکدیگر **عمود** باشند.

> [!NOTE] شرط تعامد در مختصات دکارتی
> اگر شیب مماس بر منحنی اول $m_1 = \frac{dy}{dx}$ و شیب مماس بر منحنی دوم $m_2$ باشد، شرط تعامد عبارت است از:
> $$m_1 \cdot m_2 = -1 \implies m_2 = -\frac{1}{m_1}$$

### کاربردهای مهندسی و فیزیکی
1. **خطوط میدان الکتریکی و خطوط هم‌پتانسیل:** خطوط نیرو همواره بر سطوح هم‌پتانسیل عمودند.
2. **جریان شاره‌ها (سیالات):** خطوط جریان (Streamlines) بر خطوط پتانسیل سرعت عمود هستند.
3. **انتقال حرارت:** خطوط شار گرما بر منحنی‌های همدما (Isotherms) تعامد دارند.

## الگوریتم گام‌به‌گام محاسبه در مختصات دکارتی
برای یافتن مسیرهای متعامد خانواده منحنی‌های $F(x, y, c) = 0$:

1. **مشتق‌گیری:** نسبت به $x$ مشتق ضمنی بگیرید تا عبارتی شامل $x$، $y$، $y'$ و پارامتر $c$ به دست آید.
2. **حذف پارامتر $c$:** با ترکیب معادله اصلی و معادله مشتق، پارامتر $c$ را حذف کنید تا معادله دیفرانسیل خانواده اصلی حاصل شود:
   $$\frac{dy}{dx} = f(x, y)$$
3. **جایگزینی شرط تعامد:** در معادله دیفرانسیل به دست آمده، $y'$ را با $-\frac{1}{y'}$ جایگزین کنید:
   $$\frac{dy}{dx} = -\frac{1}{f(x, y)}$$
4. **حل معادله دیفرانسیل جدید:** با انتگرال‌گیری از این معادله، خانواده مسیرهای متعامد به دست می‌آید.

---

## الگوریتم محاسبه در مختصات قطبی
اگر خانواده منحنی‌ها در مختصات قطبی به صورت $F(r, \theta, c) = 0$ داده شده باشند:

1. نسبت به $\theta$ مشتق بگیرید و پارامتر $c$ را حذف کنید تا به رابطه زیر برسید:
   $$\frac{dr}{d\theta} = f(r, \theta)$$
2. برای به دست آوردن معادله دیفرانسیل مسیرهای متعامد، جایگزینی زیر را انجام دهید:
   $$\frac{dr}{d\theta} \longleftarrow -r^2 \frac{d\theta}{dr}$$
3. معادله جداشدنی حاصل را حل کنید.

## مثال‌های کاربردی و حل‌شده

### مثال ۱: خانواده سهمی‌ها (مختصات دکارتی)
مسیرهای متعامد خانواده سهمی‌های $y = c x^2$ را بیابید.

**پاسخ:**
1. مشتق نسبت به $x$:
   $$y' = 2 c x$$
2. حذف $c$ از رابطه اصلی ($c = \frac{y}{x^2}$):
   $$y' = 2 \left(\frac{y}{x^2}\right) x = \frac{2y}{x}$$
3. اعمال شرط تعامد ($y' \to -\frac{1}{y'}$):
   $$-\frac{1}{y'} = \frac{2y}{x} \implies y' = -\frac{x}{2y} \implies \frac{dy}{dx} = -\frac{x}{2y}$$
4. جداسازی متغیرها و انتگرال‌گیری:
   $$2y \, dy = -x \, dx \implies \int 2y \, dy = -\int x \, dx$$
   $$y^2 = -\frac{x^2}{2} + k \implies \frac{x^2}{2} + y^2 = k \implies \frac{x^2}{2k} + \frac{y^2}{k} = 1$$

> [!TIP] نتیجه هندسی
> مسیرهای متعامد خانواده **سهمی‌ها** ($y = cx^2$)، خانواده **بیضی‌هایی** به مرکز مبدأ مختصات هستند.

---

### مثال ۲: دایره‌های مماس بر محورها (مختصات قطبی)
مسیرهای متعامد خانواده دایره‌های $r = c \cos\theta$ را بیابید.

**پاسخ:**
1. مشتق نسبت به $\theta$:
   $$\frac{dr}{d\theta} = -c \sin\theta$$
2. از رابطه اصلی داریم $c = \frac{r}{\cos\theta}$، پس:
   $$\frac{dr}{d\theta} = -\left(\frac{r}{\cos\theta}\right)\sin\theta = -r \tan\theta$$
3. جایگزینی شرط تعامد قطبی ($\frac{dr}{d\theta} \to -r^2 \frac{d\theta}{dr}$):
   $$-r^2 \frac{d\theta}{dr} = -r \tan\theta \implies r \frac{d\theta}{dr} = \tan\theta$$
4. تفکیک متغیرها:
   $$\frac{dr}{r} = \frac{d\theta}{\tan\theta} = \cot\theta \, d\theta$$
   $$\ln|r| = \ln|\sin\theta| + \ln|k| \implies r = k \sin\theta$$

> مسیرهای متعامد دایره‌های مماس بر محور عمودی ($r = c\cos\theta$)، دایره‌های مماس بر محور افقی ($r = k\sin\theta$) هستند.
